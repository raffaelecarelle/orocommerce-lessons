Import ed export

services:
    algoritma_training.importexport.data_converter:
        parent: oro_importexport.data_converter.configurable

    algoritma_training.importexport.strategy:
        parent: oro_importexport.strategy.configurable_add_or_replace

    algoritma_training.importexport.processor.import:
        parent: oro_importexport.processor.import_abstract
        calls:
            - [setDataConverter, ['@algoritma_training.importexport.data_converter']]
            - [setStrategy, ['@algoritma_training.importexport.strategy']]
        tags:
            - { name: oro_importexport.processor, type: import, entity: 'Algoritma\Bundle\TrainingBundle\Entity\ErpMapping', alias: algoritma_erp_mapping.add_or_replace }
            - { name: oro_importexport.processor, type: import_validation, entity: 'Algoritma\Bundle\TrainingBundle\Entity\ErpMapping', alias: algoritma_erp_mapping.add_or_replace }

    algoritma_training.importexport.processor.export:
        parent: oro_importexport.processor.export_abstract
        calls:
            - [setDataConverter, ['@algoritma_training.importexport.data_converter']]
        tags:
            - { name: oro_importexport.processor, type: export, entity: 'Algoritma\Bundle\TrainingBundle\Entity\ErpMapping', alias: algoritma_erp_mapping }
TipoServe per
importL'import vero
import_validationL'anteprima di validazione (dichiaralo sempre insieme a import)
exportL'esportazione dei dati
export_templateIl file di esempio scaricabile
 
# idempotenza: il campo chiave si marca sull'entità
#[ConfigField(defaultValues: ['importexport' => ['identity' => true]])]

# il file deve stare DENTRO la cartella del progetto: il comando gira nel container
# --email è obbligatorio; il job della validazione è diverso da quello dell'import

orobox console oro:import:file var/data.csv \
  --jobName=entity_import_validation_from_csv \
  --processor=algoritma_erp_mapping.add_or_replace \
  --email=admin@example.com --validation              # prima sempre in validazione

orobox console oro:import:file var/data.csv \
  --jobName=entity_import_from_csv \
  --processor=algoritma_erp_mapping.add_or_replace \
  --email=admin@example.com

# il comando SCHEDULA: senza consumer attivo non succede nulla
orobox console oro:message-queue:consume

orobox console debug:container --tag=oro_importexport.processor   # cosa è registrato

Job batch generici del core: entity_import_from_csv, entity_export_to_csv (in vendor/oro/platform/src/Oro/Bundle/ImportExportBundle/Resources/config/batch_jobs.yml).

Message queue

# Topic: contratto del payload
class SyncErpMappingTopic extends AbstractTopic
{
    public static function getName(): string { return 'algoritma.training.sync_erp_mapping'; }
    public static function getDescription(): string { return '...'; }

    public function configureMessageBody(OptionsResolver $resolver): void
    {
        $resolver->setRequired('mappingId')->setAllowedTypes('mappingId', 'int');
        $resolver->setDefault('force', false)->setAllowedTypes('force', 'bool');
    }
}

# Processor: MessageProcessorInterface + TopicSubscriberInterface
public static function getSubscribedTopics(): array { return [SyncErpMappingTopic::getName()]; }
public function process(MessageInterface $message, SessionInterface $session): string
{
    // self::ACK | self::REJECT | self::REQUEUE
}

# Producer
$this->producer->send(SyncErpMappingTopic::getName(), ['mappingId' => 42]);
# argomento nel services.yml: '@oro_message_queue.client.message_producer'
# mq_topics.yml
services:
    _defaults:
        tags: [ { name: oro_message_queue.topic } ]
    Algoritma\Bundle\TrainingBundle\Async\Topic\SyncErpMappingTopic: ~

# mq_processors.yml
services:
    _instanceof:
        Oro\Component\MessageQueue\Client\TopicSubscriberInterface:
            tags: [ { name: oro_message_queue.client.message_processor } ]
    Algoritma\Bundle\TrainingBundle\Async\SyncErpMappingProcessor:
        arguments: [ '@doctrine', '@logger' ]
EsitoQuando
ACKLavoro completato
REJECTRiprovare non può funzionare (payload invalido, record inesistente)
REQUEUEProblema temporaneo (sistema esterno giù, lock occupato)
 
orobox console oro:message-queue:consume -vv      # riavvialo dopo ogni modifica al codice
orobox console oro:message-queue:topics
orobox console oro:message-queue:destinations
orobox console oro:message-queue:create-queues

In Community Edition il trasporto è DBAL (coda su PostgreSQL, con polling): qualche secondo di latenza è normale. In Enterprise è RabbitMQ; il codice non cambia.

Errori frequenti

SintomoCausa
Modifica ad api.yml senza effettoManca oro:api:cache:clear
Risorsa API 404 con alias «ovvio»L'alias reale lo dà oro:api:dump: non indovinarlo
API 403 con token validoACL o ownership dell'utente collegato al client OAuth
Import che duplicaManca importexport.identity sul campo chiave
Import «non trova le colonne»Intestazioni del CSV diverse da quelle attese: scarica il template
File not found con un percorso validoIl file è fuori dalla cartella del progetto: il comando gira nel container
The --email option is requiredoro:import:file vuole sempre --email
Import «scheduled» ma nessun datoIl consumer non sta girando: l'import passa dalla coda
Messaggio pubblicato ma mai eseguitoConsumer non in esecuzione, o processor senza tag
Il processor usa codice vecchioIl consumer è un processo lungo: riavvialo
Ciclo infinito di riconsumoREQUEUE restituito su un errore permanente
